Introductions

A step-by-step demonstration of the application of the developed model. Two case studies are presented. In the first, imipramine and ribitol retention factors are predicted based on desipramine measurements, allowing assessment of the predictive contribution of structural similarity, which is relevant for imipramine (similar) but not for ribitol (dissimilar). In the second case study, imipramine and ribitol retention factors are predicted based on desipramine measurements together with a single ribitol and imipramine measurement. This analysis evaluates the additional predictive value provided by incorporating an individual experimental measurement.

Stan

Initialize variables and parameters

Code 
nObs <- length(data$ID)
nAnalytes <- length(unique(data$ID))
start <- (1:nObs)[!duplicated(data$ID)]
end <- c(start[-1] - 1, nObs)
nK = ncol(functional_groups)

init_aprox <- data %>%
  group_by(ID) %>%
  summarise(
    S1 = if_else(n() == 1,-16,polyfit(fi / (1 + 2 * fi), logk, 1)[1]),
    logkw = if_else(n() == 1,polyfit(-16 * fi / (1 + 2 * fi), logk, 0), 
                             polyfit(fi / (1 + 2 * fi), logk, 1)[2]),
    S2 = 2) %>%
  ungroup() %>%
  select(logkw, S2, S1) %>%
  mutate(S1 = -S1 / (1 + 2))

idxcv = which(data$ID==3) # imipramine

datastruct <- with(data,
                  list(nAnalytes=length(unique(data$ID)),
                       nObs=length(data$ID),
                       analyte=match(data$ID, unique(data$ID)),
                       fi=data$fi,
                       logPobs=descriptor_df$LogP,
                       nK = nK,
                       fgrp = functional_groups,
                       similarity_x = similarity_matrix,
                       start = start,
                       end= end,
                       nObscv = length(idxcv),
                       idxcv = idxcv,
                       logkobs=logk,
                       run_estimation=1))

# initialize the values for each variable in each chain:
init <- function(){
  list(param = cbind(init_aprox$logkw,init_aprox$S1),
       pilogkw = rep(0,nK),
       piS1 = rep(0,nK)
  )
}

Case study 1:

QSRR and similarity matrix for imipramine and ribitol. The data for desipramine used for predictions.

The green area represents population predictions for future observations based solely on population-level parameters and predictors, without any individual-level data. The purple area represents individual predictions, conditional additionally on the retention measurements presented in red. All observation are presented as closed white symbols.

Code 
p <- fit$draws()%>% 
  as_draws_df()%>%
  slice_sample(n=1000) %>%
  tidybayes::spread_draws(param[i,..],param_pop[i,..]) %>%
  rename(ID=i) %>%
  left_join(analytes_names) %>%
  tidyr::expand_grid(fi = seq(0,1,0.1)) %>%
  mutate(ind = param.1-param.2*(1+2.90)*fi/(1+2.90*fi)) %>%
  mutate(pop = param_pop.1-param_pop.2*(1+2.90)*fi/(1+2.90*fi)) %>%
  select("ID","Analyte",".chain",".iteration",".draw","fi","pop","ind") %>%
    pivot_longer(cols = c(pop,ind),
                 names_to = "type",
                 values_to = "logk") %>%
  ggplot(aes(x = fi, y=logk, color = type, fill = type)) +
  ggdist::stat_lineribbon(.width = c(.90), alpha = 1/2) +
  geom_point(data = data%>%mutate(type="obs"),aes(x=fi, y=logk, color=type), 
                                                   shape = 21,  
                                                   fill = "white",size = 2, stroke = 0.8) + 
 geom_point(data = data[idxcv,]%>%mutate(type="obs"),aes(x=fi, y=logk, color=type), 
                                                   shape = 21,  
                                                   fill = "red",size = 2, stroke = 0.8) + 
  facet_wrap(~Analyte, ncol = 1)+

  theme(legend.position = "none") +
  coord_cartesian(xlim=c(0,1),ylim=c(-1,2))+

# Introductions

A step-by-step demonstration of the application of the developed model. Two case studies are presented. In the first, imipramine and ribitol retention factors are predicted based on desipramine measurements, allowing assessment of the predictive contribution of structural similarity, which is relevant for imipramine (similar) but not for ribitol (dissimilar). In the second case study, imipramine and ribitol retention factors are predicted based on desipramine measurements together with a single ribitol and imipramine measurement. This analysis evaluates the additional predictive value provided by incorporating an individual experimental measurement.

# Initialize variables and parameters
```{r stan-setup}
nObs <- length(data$ID)
nAnalytes <- length(unique(data$ID))
start <- (1:nObs)[!duplicated(data$ID)]
end <- c(start[-1] - 1, nObs)
nK = ncol(functional_groups)

init_aprox <- data %>%
  group_by(ID) %>%
  summarise(
    S1 = if_else(n() == 1,-16,polyfit(fi / (1 + 2 * fi), logk, 1)[1]),
    logkw = if_else(n() == 1,polyfit(-16 * fi / (1 + 2 * fi), logk, 0), 
                             polyfit(fi / (1 + 2 * fi), logk, 1)[2]),
    S2 = 2) %>%
  ungroup() %>%
  select(logkw, S2, S1) %>%
  mutate(S1 = -S1 / (1 + 2))

idxcv = which(data$ID==3) # imipramine

datastruct <- with(data,
                  list(nAnalytes=length(unique(data$ID)),
                       nObs=length(data$ID),
                       analyte=match(data$ID, unique(data$ID)),
                       fi=data$fi,
                       logPobs=descriptor_df$LogP,
                       nK = nK,
                       fgrp = functional_groups,
                       similarity_x = similarity_matrix,
                       start = start,
                       end= end,
                       nObscv = length(idxcv),
                       idxcv = idxcv,
                       logkobs=logk,
                       run_estimation=1))

# initialize the values for each variable in each chain:
init <- function(){
  list(param = cbind(init_aprox$logkw,init_aprox$S1),
       pilogkw = rep(0,nK),
       piS1 = rep(0,nK)
  )
}
```
